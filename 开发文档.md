```sql
CREATE TABLE user (
    id          BIGINT PRIMARY KEY AUTO_INCREMENT,

    FOREIGN KEY (user_id)    REFERENCES user(id),
    FOREIGN KEY (product_id) REFERENCES product(id)

-- 订单表
CREATE TABLE `order` (
    id          BIGINT PRIMARY KEY AUTO_INCREMENT,
```

### 4.4 JWT 工具类

```kotlin
@Component
class JwtUtil(
    @Value("\${jwt.secret}") private val secret: String,
    @Value("\${jwt.expiration}") private val expiration: Long
) {
    private val key by lazy { Keys.hmacShaKeyFor(secret.toByteArray()) }

    fun generateToken(userId: Long, role: Int): String =
        Jwts.builder()
            .subject(userId.toString())
            .claim("role", role)
            .issuedAt(Date())
            .expiration(Date(System.currentTimeMillis() + expiration))
            .signWith(key)
            .compact()

    fun parseToken(token: String): Claims =
        Jwts.parser().verifyWith(key).build()
            .parseSignedClaims(token).payload
}
```

### 4.5 JWT 拦截器

```kotlin
@Component
class JwtInterceptor(private val jwtUtil: JwtUtil) : HandlerInterceptor {

    override fun preHandle(request: HttpServletRequest, response: HttpServletResponse, handler: Any): Boolean {
        val token = request.getHeader("Authorization")?.removePrefix("Bearer ")
            ?: throw BizException(401, "未登录")

        return try {
            val claims = jwtUtil.parseToken(token)
            request.setAttribute("userId", claims.subject.toLong())
            request.setAttribute("role", claims["role"] as Int)
            true
        } catch (e: Exception) {
            throw BizException(401, "Token 无效或已过期")
        }
    }
}
```

### 4.6 全局异常处理

```kotlin
class BizException(val code: Int = 400, override val message: String) : RuntimeException(message)

@RestControllerAdvice
class GlobalExceptionHandler {

    @ExceptionHandler(BizException::class)
    fun handleBiz(e: BizException): Result<Nothing> =
        Result.error(e.code, e.message)

    @ExceptionHandler(MethodArgumentNotValidException::class)
    fun handleValidation(e: MethodArgumentNotValidException): Result<Nothing> =
        Result.error(400, e.bindingResult.allErrors.first().defaultMessage ?: "参数校验失败")

    @ExceptionHandler(Exception::class)
    fun handleException(e: Exception): Result<Nothing> =
        Result.error(500, "服务器内部错误")
}
```

### 6.2 Axios 请求封装

```typescript
import axios from 'axios'
import { ElMessage } from 'element-plus'
import router from '@/router'

const request = axios.create({
  baseURL: '/api',
  timeout: 10000,
})

// 请求拦截：自动携带 Token
request.interceptors.request.use((config) => {
  const token = localStorage.getItem('token')
  if (token) {
    config.headers.Authorization = `Bearer ${token}`
  }
  return config
})

// 响应拦截：统一错误处理
request.interceptors.response.use(
  (res) => {
    if (res.data.code !== 200) {
      ElMessage.error(res.data.message)
      return Promise.reject(res.data)
    }
    return res.data
  },
  (error) => {
    if (error.response?.status === 401) {
      localStorage.removeItem('token')
      router.push('/login')
    }
    ElMessage.error('网络请求失败')
    return Promise.reject(error)
  }
)

export default request
```

### 6.3 路由守卫

```typescript
router.beforeEach((to, from, next) => {
  const token = localStorage.getItem('token')
  const whiteList = ['/login', '/register']

  if (whiteList.includes(to.path)) {
    next()
  } else if (!token) {
    next('/login')
  } else {
    next()
  }
})
```

## 七、核心业务流程

### 7.1 下单流程

```
1. 用户选择购物车中的商品 → 点击"去结算"
2. 前端调用 POST /api/orders，传入 addressId + 商品列表
3. 后端校验库存、计算总价、生成订单号（时间戳+随机数）
4. 创建 order 和 order_item 记录，状态设为 0（待接单）
5. 清除购物车中对应的商品
6. 返回订单信息给前端，跳转到"模拟支付"页面
7. 用户点击"确认支付"按钮，前端直接视为支付成功
```

### 7.2 商家接单与配送流程

```
1. 商家在订单管理页看到新订单，点击"接单"
2. 后端将订单状态改为 1（备货中）
3. 商家备货完成后点击"备货完成"
4. 后端自动分配一台空闲无人机（status=0），将其状态改为 1（配送中）
5. 订单状态改为 2（配送中），记录 drone_id
6. 同时在 drone_log 表插入一条配送记录
7. 用户确认收货后，订单状态改为 3（已送达）
8. 无人机状态恢复为 0（空闲），drone_log 记录 end_time
```

### 7.3 客服聊天流程

```
1. 用户进入客服页面，点击"发起咨询"（可选择关联订单）
2. 前端调用 POST /api/chat/sessions 创建会话
3. 前端通过 WebSocket /ws/chat 建立连接
4. 用户发送消息 → WebSocket 推送给管理员
5. 管理员在客服工作台看到会话，回复消息 → WebSocket 推送给用户
6. 管理员处理完毕后点击"关闭会话"
```
